357            if (empty($table['Engine'])):
358                continue;
359            endif;
360
361            /* Calcolo della dimensione totale (Dati + Indici) convertita in MB */
362            $totalSizeBytes = $table['Data_length'] + $table['Index_length'];
363            $sizeMB = round($totalSizeBytes / 1048576, 2);
364
365            /* Calcolo dell'overhead (Spazio liberabile/frammentato) convertito in MB */
366            $overheadBytes = $table['Data_free'];
367            $overheadMB = round($overheadBytes / 1048576, 2);
368
369            $result[] = [
370                'name' => $table['Name'],
371                'rows' => $table['Rows'] ?? 0,
372                'size' => $sizeMB,
373                'overhead' => $overheadMB,
374                'engine' => $table['Engine']
375            ];
376        }
377
378        return $result;
379    }

381    /**
382     * Avvia la manutenzione fisica del database (Deframmentazione).
383     * 
384     * Accetta una o più tabelle ed esegue in sequenza i comandi nativi MySQL: 
385     * ANALYZE (statistiche), CHECK (integrità) e OPTIMIZE (recupero spazio frammentato). 
386     * Una volta finito, restituisce lo stato aggiornato (pesi in MB) da mostrare all'utente.
387     *
388     * @param string|array $target Il nome della singola tabella o l'array di tutte le tabelle da ottimizzare
389     * @return array|bool Lo stato aggiornato della/e tabella/e elaborata/e
390     */
391    public function runOptimization(string|array $target): array|bool
392    {
393        /* Normalizziamo l'input in un array per processare le query */
394        $tables = array_values(array_unique(is_array($target) ? $target : [$target]));
395        $availableTables = array_column($this->getTablesStatus(), 'name');
396
397        if ($tables === [] || array_diff($tables, $availableTables) !== []):
398            return false;
399        endif;
400
401        try {
402            foreach ($tables as $table):
403                $escapedTable = $this->db->escapeIdentifiers($table);
404
405                foreach (['analyze', 'check', 'optimize'] as $operation):
406                    $query = $this->db->query("{$operation} table {$escapedTable}");
407
408                    if ( ! $this->maintenanceQuerySucceeded($query)):
409                        return false;
410                    endif;
411                endforeach;
412            endforeach;
413        } catch (\Throwable $e) {
414            log_message('error', 'Tools runOptimization error: ' . $e->getMessage());
415            return false;
416        }
417
418        /* Registrazione attività */
419        $currentAdmin = service('authorization')->currentAdmin();
420        $targetLog = is_string($target) ? "la tabella {$target}" : "tutte le tabelle del database";
421        log_admin_activity('OPTIMIZE_DB', 'tools', sprintf(lang('backend/tools.audits.targetLog'), $targetLog), $currentAdmin);
422
423        /* Se la richiesta era per una tabella singola, estraiamo solo quella */
424        if (is_string($target)):
425            return $this->getTablesStatus($target);
426        endif;
427
428        /* Se la richiesta era un array (tutte le tabelle), restituiamo l'intero database aggiornato */
429        return $this->getTablesStatus();
430    }

529    /**

554    /**
555     * Genera un backup completo del database (struttura e dati) salvandolo su disco.
556     * 
557     * Operazione massiva: cicla tutte le tabelle, genera le query per ricrearle (CREATE TABLE) 
558     * e per reinserire i dati (INSERT INTO), proteggendo stringhe e caratteri speciali.
559     * Salva tutto in un file `.sql`, lo comprime in uno `.zip` per risparmiare spazio, 
560     * cancella l'originale in chiaro ed elimina eventuali backup vecchi mantenendo solo gli ultimi 10.
561     *
562     * @return bool True se il file zip è stato creato con successo, false altrimenti
563     */
564    public function generateDatabaseBackups(): bool
565    {
566        $path = WRITEPATH . 'backups/database/';
567
568        if ( ! is_dir($path) && ( ! mkdir($path, 0775, true) && ! is_dir($path))):
569            log_message('error', 'Tools backup error: impossibile creare la directory ' . $path);
570            return false;
571        endif;
572
573        $date = date('Y-m-d_H-i-s');
574        $sqlFilename = 'backup_' . $date . '.sql';
575        $zipFilename = 'backup_' . $date . '.zip';
576
577        $sqlPath = $path . $sqlFilename;
578        $zipPath = $path . $zipFilename;
579
580        if (file_exists($sqlPath) || file_exists($zipPath)):
581            log_message('error', 'Tools backup error: esiste già un file con timestamp ' . $date);
582            return false;
583        endif;
584
585        $fileHandler = null;
586        $snapshotStarted = false;
587        $backupCompleted = false;
588
589        try {
590            $fileHandler = fopen($sqlPath, 'xb');
591
592            if ($fileHandler === false):
593                throw new \RuntimeException('Impossibile creare il file SQL temporaneo.');
594            endif;
595
596            if ($this->db->query('set transaction isolation level repeatable read') === false
597                || $this->db->query('start transaction with consistent snapshot') === false):
598                throw new \RuntimeException('Impossibile avviare lo snapshot consistente del database.');
599            endif;
600
601            $snapshotStarted = true;
602
603            $this->writeBackupData($fileHandler, '/* Backup database generato il ' . date('Y-m-d H:i:s') . " */\n\n");
604            $this->writeBackupData($fileHandler, "set foreign_key_checks = 0;\n\n");
605
606            $tables = $this->db->listTables();
607            $views = [];
608
609            foreach ($tables as $table):
610                $escapedTable = $this->db->escapeIdentifiers($table);
611                $schemaQuery = $this->db->query("show create table {$escapedTable}");
612
613                if ($schemaQuery === false):
614                    throw new \RuntimeException('Impossibile leggere la struttura di ' . $table . '.');
615                endif;
616
617                $schemaRow = $schemaQuery->getRowArray();
618                $createTable = $schemaRow['Create Table'] ?? null;
619                $createView = $schemaRow['Create View'] ?? null;
620
621                if (is_string($createView) && $createView !== ''):
622                    $views[] = ['name' => $escapedTable, 'sql' => $createView];
623                    continue;
624                endif;
625
626                if ( ! is_string($createTable) || $createTable === ''):
627                    throw new \RuntimeException('Definizione SQL non disponibile per ' . $table . '.');
628                endif;
629
630                $fieldNames = $this->db->getFieldNames($table);
631
632                if ($fieldNames === []):
633                    throw new \RuntimeException('Nessuna colonna disponibile per ' . $table . '.');
634                endif;
635
636                $escapedFields = array_map(fn($field) => $this->db->escapeIdentifiers($field), $fieldNames);
637                $fieldList = implode(', ', $escapedFields);
638
639                $this->writeBackupData($fileHandler, "drop table if exists {$escapedTable};\n");
640                $this->writeBackupData($fileHandler, $createTable . ";\n\n");
641
642                $dataQuery = $this->db->query("select {$fieldList} from {$escapedTable}");
643
644                if ($dataQuery === false):
645                    throw new \RuntimeException('Impossibile leggere i dati di ' . $table . '.');
646                endif;
647
648                while (($dataRow = $dataQuery->getUnbufferedRow('array')) !== null):
649                    $values = [];
650
651                    foreach ($dataRow as $value):
652                        $values[] = $value === null ? 'null' : $this->db->escape($value);
653                    endforeach;
654
655                    $this->writeBackupData(
656                        $fileHandler,
657                        "insert into {$escapedTable} ({$fieldList}) values (" . implode(', ', $values) . ");\n"
658                    );
659                endwhile;
660
661                $dataQuery->freeResult();
662                $this->writeBackupData($fileHandler, "\n");
663            endforeach;
664
665            foreach ($views as $view):
666                $this->writeBackupData($fileHandler, "drop view if exists {$view['name']};\n");
667                $this->writeBackupData($fileHandler, $view['sql'] . ";\n\n");
668            endforeach;
669
670            $this->writeBackupData($fileHandler, "set foreign_key_checks = 1;\n");
671
672            if ($this->db->query('rollback') === false):
673                throw new \RuntimeException('Impossibile chiudere lo snapshot del database.');
674            endif;
675
676            $snapshotStarted = false;
677
678            if ( ! fclose($fileHandler)):
679                throw new \RuntimeException('Impossibile chiudere il file SQL temporaneo.');
680            endif;
681
682            $fileHandler = null;
683
684            $zip = new \ZipArchive();
685
686            if ($zip->open($zipPath, \ZipArchive::CREATE | \ZipArchive::OVERWRITE) !== true):
687                throw new \RuntimeException('Impossibile creare l’archivio ZIP.');
688            endif;
689
690            $fileAdded = $zip->addFile($sqlPath, $sqlFilename);
691            $zipClosed = $zip->close();
692
693            if ( ! $fileAdded || ! $zipClosed):
694                throw new \RuntimeException('Impossibile completare l’archivio ZIP.');
695            endif;
696
697            $verificationZip = new \ZipArchive();
698
699            if ($verificationZip->open($zipPath) !== true):
700                throw new \RuntimeException('L’archivio ZIP generato non è leggibile.');
701            endif;
702
703            $archiveContainsDump = $verificationZip->locateName($sqlFilename) !== false;
704            $verificationZip->close();
705            $zipSize = filesize($zipPath);
706
707            if ( ! $archiveContainsDump || $zipSize === false || $zipSize === 0):
708                throw new \RuntimeException('L’archivio ZIP generato non contiene il dump SQL.');
709            endif;
710
711            if ( ! unlink($sqlPath)):
712                throw new \RuntimeException('Impossibile eliminare il file SQL temporaneo.');
713            endif;
714
715            $backupCompleted = true;
716        } catch (\Throwable $e) {
717            log_message('error', 'Tools generateDatabaseBackups error: ' . $e->getMessage());
718            return false;
719        } finally {
720            if ($snapshotStarted):
721                try {
722                    $this->db->query('rollback');
723                } catch (\Throwable $e) {
724                    log_message('error', 'Tools backup rollback error: ' . $e->getMessage());
725                }
726            endif;
727
728            if (is_resource($fileHandler)):
729                fclose($fileHandler);
730            endif;
731
732            if (file_exists($sqlPath) && ! unlink($sqlPath)):
733                log_message('error', 'Tools backup cleanup error: impossibile eliminare ' . $sqlPath);
734            endif;
735
736            if ( ! $backupCompleted && file_exists($zipPath) && ! unlink($zipPath)):
737                log_message('error', 'Tools backup cleanup error: impossibile eliminare ' . $zipPath);
738            endif;
739        }
740
741        $files = array_values(array_filter(glob($path . '*.zip') ?: [], function($file) {
742            return preg_match(self::BACKUP_FILENAME_PATTERN, basename($file)) === 1;
743        }));
744
745        if (count($files) > 10):
746            usort($files, function($a, $b) {
747                return filemtime($a) <=> filemtime($b);
748            });
749
750            $filesToDelete = count($files) - 10;
751
752            for ($i = 0; $i < $filesToDelete; $i++):
753                if (file_exists($files[$i]) && ! unlink($files[$i])):
754                    log_message('error', 'Tools backup rotation error: impossibile eliminare ' . $files[$i]);
755                endif;
756            endfor;
757        endif;
758
759        try {
760            $currentAdmin = service('authorization')->currentAdmin();
761            log_admin_activity('GENERATE_BACKUP', 'tools', sprintf(lang('backend/tools.audits.generateBackup'), $zipFilename), $currentAdmin);
762        } catch (\Throwable $e) {
763            log_message('error', 'Tools backup audit error: ' . $e->getMessage());
764        }
765
766        return true;
767    }

769    /**
770     * Scrive integralmente un blocco nel dump, intercettando scritture parziali o fallite.
771     */
772    private function writeBackupData($handle, string $data): void
773    {
774        $length = strlen($data);
775        $written = 0;
776
777        while ($written < $length):
778            $result = fwrite($handle, substr($data, $written));
779
780            if ($result === false || $result === 0):
781                throw new \RuntimeException('Scrittura incompleta del dump SQL.');
782            endif;
783
784            $written += $result;
785        endwhile;
786    }

792     * ignorando eventuali percorsi completi. Questo blocca attacchi di "Path Traversal" 
793     * (tentativi di cancellare file fuori dalla cartella consentita usando stringhe come "../").
794     *
795     * @param string $filename Il nome del file zip da rimuovere
796     * @return bool True se eliminato, false se il file non esiste o l'operazione fallisce

801            return false;
802        endif;
803
804        /* basename protegge il percorso assicurando che sia solo il nome del file */
805        $path = WRITEPATH . 'backups/database/' . basename($filename);
806

815            endif;
816        endif;

821    /**
